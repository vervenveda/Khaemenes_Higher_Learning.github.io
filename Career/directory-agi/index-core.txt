<script>
(() => {
  const $=id=>document.getElementById(id);
  const esc=v=>String(v??"").replace(/[&<>"']/g,ch=>({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[ch]));

  function renderNotes(){
    const rows=CareerDirectoryAGI.notes();
    $("noteList").innerHTML=rows.length?rows.slice().reverse().map(n=>`<li><small>${esc(n.createdAt)}</small><br>${esc(n.text)}</li>`).join(""):"<li>No notes.</li>";
  }
  function render(term=""){
    const rows=term?CareerDirectoryAGI.find(term):CareerDirectoryAGI.all();
    const byCat=new Map();
    rows.forEach(r=>{if(!byCat.has(r.category))byCat.set(r.category,[]);byCat.get(r.category).push(r)});
    $("list").innerHTML=[...byCat.entries()].map(([cat,items])=>`
      <details>
        <summary>${esc(cat)} · ${items.length}</summary>
        <div class="group-body">
          ${items.map(r=>`
            <div class="item">
              <strong>${esc(r.title)}</strong>
              <span class="small">${esc(r.description||r.repository||"")}</span><br>
              <button data-id="${esc(r.id)}">Open in Career Portal</button>
            </div>`).join("")}
        </div>
      </details>`).join("")||"<p>No matches.</p>";
    $("list").querySelectorAll("[data-id]").forEach(btn=>btn.addEventListener("click",()=>{
      const r=CareerDirectoryAGI.all().find(x=>x.id===btn.dataset.id);
      if(!r)return;
      if(window.parent && window.parent!==window){
        window.parent.dispatchEvent(new CustomEvent("career-directory-launch",{detail:{resource:r}}));
      }else{
        location.href=r.url;
      }
    }));
    const s=CareerDirectoryAGI.summary();
    $("stats").innerHTML=`<div class="stat"><b>${s.records}</b><span>Records</span></div><div class="stat"><b>${s.categories}</b><span>Categories</span></div><div class="stat"><b>${s.validationIssues}</b><span>Issues</span></div>`;
    $("status").textContent=s.error?`Fallback · ${s.error}`:`Federation ready · Directory AGI v${s.version}`;
  }
  function command(){
    const raw=$("command").value.trim(),lower=raw.toLowerCase();let out="";
    if(!raw)return;
    if(lower==="help")out="find <term>\\nresolve <name>\\nshow category <name>\\nstats\\nvalidate\\nhelp";
    else if(lower==="stats")out=JSON.stringify(CareerDirectoryAGI.summary(),null,2);
    else if(lower==="validate"){const p=CareerDirectoryAGI.validate();out=p.length?p.join("\\n"):"Validation passed."}
    else if(lower.startsWith("find ")){const r=CareerDirectoryAGI.find(raw.slice(5)).slice(0,25);out=r.length?r.map(x=>`${x.category} · ${x.title}\\n${x.url}`).join("\\n\\n"):"No matches."}
    else if(lower.startsWith("resolve ")){const r=CareerDirectoryAGI.resolve(raw.slice(8));out=r?`${r.title}\\n${r.category}\\n${r.url}`:"No match."}
    else if(lower.startsWith("show category ")){const q=raw.slice(14).toLowerCase(),r=CareerDirectoryAGI.all().filter(x=>x.category.toLowerCase().includes(q));out=r.length?r.map(x=>`${x.title} · ${x.url}`).join("\\n"):"Category not found."}
    else{const r=CareerDirectoryAGI.resolve(raw);out=r?`${r.title}\\n${r.url}`:"Command not recognized. Type help."}
    $("output").textContent=out;$("command").value="";
  }

  $("noteStore").addEventListener("click",()=>{if(CareerDirectoryAGI.note($("noteInput").value)){$("noteInput").value="";renderNotes()}});
  $("noteClear").addEventListener("click",()=>{CareerDirectoryAGI.clearNotes();renderNotes()});
  $("search").addEventListener("input",e=>render(e.target.value));
  $("run").addEventListener("click",command);
  $("command").addEventListener("keydown",e=>{if(e.key==="Enter")command()});
  window.addEventListener("career-directory-ready",()=>{render($("search").value);renderNotes();$("output").textContent="Career Directory AGI ready. Type help."});
  if(window.CareerDirectoryAGI){render();renderNotes()}
})();
</script>
